**lead_parser.py** (הרצה: `python lead_parser.py leads.txt`, ב-Windows אפשר גם `py lead_parser.py leads.txt`):
```python
import re
import json
import os
import sys
import logging
import datetime
import collections
from abc import ABC, abstractmethod

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# Config for things that never change
CONFIG = {
    "SEPARATOR": "|",
    "ENCODING": "utf-8",
    "MIN_FIELDS": 4,
    "PHONE_REGEX": r"^05\d-?\d{7}$",
    "CURRENCY": "ILS",
    "CATEGORIES": {
        "מטבח": "kitchen",
        "אמבטיה": "bathroom",
        "צבע": "painting",
    },
    "DEFAULT_CATEGORY": "other",
}


class BaseParser(ABC):
    """Abstract base parser."""

    @abstractmethod
    def parse(self, line):
        """Parse a line."""
        pass


class LeadParser(BaseParser):
    """Parses a lead line."""

    def __init__(self, config):
        self.config = config

    def parse(self, line):
        # split the line by separator
        parts = line.split(self.config["SEPARATOR"])
        # strip each part
        parts = [p.strip() for p in parts]
        return parts


class ParserFactory:
    """Factory that creates parsers."""

    _instance = None

    @staticmethod
    def get_parser(kind="lead"):
        if kind == "lead":
            return LeadParser(CONFIG)
        else:
            raise ValueError("Unknown parser kind")


class Lead:
    def __init__(self, name, phone, text, budget):
        self.name = name
        self.phone = phone
        self.text = text
        self.budget = budget
        self.category = None

    def get_name(self):
        return self.name

    def get_phone(self):
        return self.phone

    def get_text(self):
        return self.text

    def get_budget(self):
        return self.budget

    def set_category(self, category):
        self.category = category

    def get_category(self):
        return self.category


class PhoneValidator:
    def validate(self, phone):
        if phone is None:
            return False
        if not isinstance(phone, str):
            return False
        if re.match(CONFIG["PHONE_REGEX"], phone):
            return True
        return False


class BudgetConverter:
    def convert(self, value):
        try:
            if value is None:
                return 0
            cleaned = value.replace(",", "")
            cleaned = cleaned.strip()
            result = int(cleaned)
            return result
        except ValueError:
            return 0
        except Exception as e:
            logger.error(e)
            return 0


class CategoryClassifier:
    def classify(self, text):
        for hebrew_word, category in CONFIG["CATEGORIES"].items():
            if hebrew_word in text:
                return category
        return CONFIG["DEFAULT_CATEGORY"]


class LeadBuilder:
    def __init__(self):
        self.validator = PhoneValidator()
        self.converter = BudgetConverter()
        self.classifier = CategoryClassifier()

    def build(self, parts):
        if len(parts) < CONFIG["MIN_FIELDS"]:
            return None
        name, phone, text, budget = parts[0], parts[1], parts[2], parts[3]
        if not self.validator.validate(phone):
            return None
        lead = Lead(name, phone, text, self.converter.convert(budget))
        lead.set_category(self.classifier.classify(lead.get_text()))
        return lead


class SummaryReporter:
    def __init__(self, leads):
        self.leads = leads

    def total_budget(self):
        total = 0
        for lead in self.leads:
            total = total + lead.get_budget()
        return total

    def count_by_category(self):
        counter = collections.Counter()
        for lead in self.leads:
            counter[lead.get_category()] += 1
        return dict(counter)

    def report(self):
        lines = []
        lines.append("Leads: " + str(len(self.leads)))
        lines.append("Total budget: " + str(self.total_budget()))
        for category, count in sorted(self.count_by_category().items()):
            lines.append(category + ": " + str(count))
        return "\n".join(lines)


def process_lines(lines):
    parser = ParserFactory.get_parser("lead")
    builder = LeadBuilder()
    leads = []
    for line in lines:
        if line is None or line.strip() == "":
            continue
        parts = parser.parse(line)
        lead = builder.build(parts)
        if lead is not None:
            leads.append(lead)
        else:
            logger.warning("Skipping invalid line")
    return leads


def main(path):
    with open(path, "r", encoding=CONFIG["ENCODING"]) as f:
        lines = f.readlines()
    leads = process_lines(lines)
    reporter = SummaryReporter(leads)
    print(reporter.report())


if __name__ == "__main__":
    main(sys.argv[1])
```

## הנחיות למרצה

**ריחות עיקריים בסקריפט ההדגמה (תרגיל 2):**
1. `BaseParser` אבסטרקטי עם מימוש אחד בלבד (`LeadParser`), ו-`ParserFactory` שיוצר רק סוג אחד (כולל `_instance` שלא בשימוש).
2. מחלקת `Lead` עם getters ו-setters מיותרים. בפייתון ניגשים לשדות ישירות, או משתמשים ב-dataclass.
3. `PhoneValidator`, `BudgetConverter`, `CategoryClassifier`: מחלקות עם מתודה אחת ובלי מצב. מספיקות פונקציות.
4. `CONFIG` עם ערכים שלא משתנים (`ENCODING`, `SEPARATOR`) וערך שלא בשימוש כלל (`CURRENCY`).
5. imports שלא בשימוש: `json`, `os`, `datetime`.
6. קוד הגנתי למצבים שלא קורים: `phone is None`, `isinstance(phone, str)`, `value is None`, ו-`except Exception` שמיותר לצד `except ValueError`.
7. הערות שחוזרות על הקוד (למשל `# split the line by separator`) ו-docstrings ריקים.
8. לוגיקה שאפשר להחליף ב-`sum(...)` וב-`Counter` ישירות, בלי מחלקת reporter.

**מה נחשב תשובה טובה:**
- הפישוט הסופי הוא בדרך כלל 40-60 שורות: פונקציות רגילות, קובץ אחד, בלי מחלקות, ואותו פלט בדיוק.
- התלמיד הריץ בדיקה **לפני** השינוי ואחריו, ועשה commit בין השלבים.
- תרגיל 4: התלמיד השווה בין בקשה כללית לממוקדת וראה שהכללית משנה יותר ממה שביקשו (לעיתים מוסיפה פיצ'רים או משנה את פורמט הפלט).
- תרגיל 5: הרפלקציה כוללת נימוק ממשי. דוגמה טובה: Claude הציע להסיר את בדיקת הטלפון, והתלמיד דחה כי היא מסננת לידים פיקטיביים (הפלט משתנה, ולכן זה שינוי פיצ'ר ולא פישוט). תלמיד שמסיים עם קוד דחוס וקשה לקריאה רק כדי להוריד שורות לא הבין את הנקודה.
- תרגיל 6: יש להבחין בין סיבוך מוצדק לסיבוך מיותר. תלמיד שלא מצא שום דבר למחוק מוזמן לבדוק שוב, אבל "לא לגעת" היא מסקנה לגיטימית אם יש לה נימוק.

**מלכודות נפוצות:**
- תלמידים מדלגים על הבדיקה ("הקוד קצר, זה בסדר"). עצרו אותם: בלי בדיקה זה לא פישוט אלא ניחוש.
- תלמידים מאשרים שינויים בלי `git diff`. בקשו מהם להסביר שורה אחת מהשינוי.
- ב-Windows, אם `python` לא מזוהה, אפשר להשתמש ב-`py`. אזהרת הלוג של שורה לא תקינה נכתבת ל-stderr ולכן לא נכנסת ל-`expected_output.txt`.

**ציון מוצע:** תרגילים 1-2: 15%, תרגיל 3: 20%, תרגיל 4: 25%, תרגיל 5: 15%, תרגיל 6: 25%.
